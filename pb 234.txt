/**
 * Definition for singly-linked list.
 * struct ListNode {
 *     int val;
 *     ListNode *next;
 *     ListNode() : val(0), next(nullptr) {}
 *     ListNode(int x) : val(x), next(nullptr) {}
 *     ListNode(int x, ListNode *next) : val(x), next(next) {}
 * };
 */
class Solution {
public:
    bool isPalindrome(ListNode* head)
    {
        stack<int> Stk;
        ListNode* temp = head;
        while (temp != NULL)
        {
            int x;
            x = temp->val;
            Stk.push(x);
            temp = temp->next;
        }
        temp = head;
        while (temp != NULL)
        {
            if (temp->val != Stk.top())
                return false;
            Stk.pop();
            temp = temp->next;
        }
        return true;
    }
};
